Napisz query, które pokaże klientów razem z segmentem:

high_value, jeśli suma ich zamówień jest większa lub równa 250,
standard, jeśli suma ich zamówień jest mniejsza niż 250,
no_orders, jeśli klient nie ma żadnego zamówienia.

select
c.customer_id,
c.customer_name,
sum(o.total_amount) as total_revenue,
case 
	when sum(o.total_amount) >= 250 then 'high_value'
	when sum(o.total_amount) < 250 then 'standard'
	else 'no_orders'
end as customer_segment
from course.customers c
left join course.orders o 
on c.customer_id = o.customer_id
group by c.customer_id, c.customer_name
order by c.customer_id;
